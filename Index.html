DEEP SEARCH & ODDS TRAP DETECTOR (+EV ENGINE) # --------------------------------------------------------- def detect_odds_trap(daznbet_odds, true_probability): implied_probability = (1.0 / daznbet_odds) * 100 ev_value = (true_probability * daznbet_odds) - 100 if implied_probability > true_probability + 10.0: is_trap = True warning = f"⚠️ BOOKMAKER ODDS TRAP: DAZNBET odds-keeda ({daznbet_odds}) waa dhiina ku jiro khattara!" else: is_trap = False warning = "✅ VALUE PICK APPROVED (+EV Validated)" return is_trap, warning, round(ev_value, 2) # --------------------------------------------------------- # 4. MONTE CARLO SIMULATION ENGINE (10,000 RUNS) # --------------------------------------------------------- def run_monte_carlo_simulation(home_xg, away_xg, iterations=10000): home_goals = np.random.poisson(home_xg, iterations) away_goals = np.random.poisson(away_xg, iterations) prob_mg_1_4 = np.sum((home_goals >= 1) & (home_goals <= 4)) / iterations prob_clean_sheet = np.sum(away_goals == 0) / iterations prob_corners_over_8_5 = 0.915 return { "multi_goals_1_4": round(prob_mg_1_4 * 100, 1), "home_clean_sheet_risk": round(prob_clean_sheet * 100, 1), "corners_over_8_5": round(prob_corners_over_8_5 * 100, 1) } # --------------------------------------------------------- # 5. RISK MANAGEMENT & KELLY STAKING ENGINE # --------------------------------------------------------- def calculate_kelly_stake(bankroll, confidence_pct, odds): b = odds - 1.0 p = confidence_pct / 100.0 q = 1.0 - p kelly_fraction = ((b * p) - q) / b # Fractional Kelly (Quarter Kelly for protection) safe_kelly = max(0.01, (kelly_fraction * 0.25)) suggested_stake = round(bankroll * safe_kelly, 2) stake_percentage = round(safe_kelly * 100, 1) return min(suggested_stake, round(bankroll * 0.05, 2)), stake_percentage # --------------------------------------------------------- # 6.
